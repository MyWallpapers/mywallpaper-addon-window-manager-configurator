cmake_minimum_required(VERSION 3.24)
project(MyWallpaperWindowManager VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_SUPPRESS_REGENERATION ON)
add_library(wm_model STATIC src/model.cpp src/wire.cpp)
target_include_directories(wm_model PUBLIC include "${CMAKE_CURRENT_BINARY_DIR}" third_party)
if(MSVC)
  target_compile_options(wm_model PUBLIC /utf-8 /W4 /permissive- /EHsc /Brepro)
else()
  target_compile_options(wm_model PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(model_tests tests/model_tests.cpp)
target_link_libraries(model_tests PRIVATE wm_model)
enable_testing()
add_test(NAME model_contract COMMAND model_tests)
if(WIN32)
  if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Native companion requires Windows x64 MSVC")
  endif()
  add_executable(companion src/main.cpp src/runtime.cpp src/overlay.cpp src/recovery.cpp)
  target_link_libraries(companion PRIVATE wm_model user32 gdi32 gdiplus dwmapi shell32 ole32 shcore)
  target_compile_definitions(companion PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_options(companion PRIVATE /Brepro)
  add_executable(window_fixture tests/window_fixture.cpp)
  target_link_libraries(window_fixture PRIVATE wm_model user32 dwmapi)
  target_compile_definitions(window_fixture PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
